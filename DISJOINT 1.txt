
#include <stdio.h>
#include <stdlib.h>

struct node
{
    struct node *rep;
    struct node *next;
    int data;
} *heads[50], *tails[50];

static int countRoot = 0;


void makeSet(int x)
{
    struct node *newNode;

    if (countRoot >= 50)
    {
        printf("\nMaximum number of sets reached\n");
        return;
    }

    newNode = (struct node *)malloc(sizeof(struct node));

    if (newNode == NULL)
    {
        printf("\nMemory allocation failed\n");
        return;
    }

    newNode->rep = newNode;
    newNode->next = NULL;
    newNode->data = x;

    heads[countRoot] = newNode;
    tails[countRoot] = newNode;
    countRoot++;
}


struct node* find(int a)
{
    int i;
    struct node *tmp;

    for (i = 0; i < countRoot; i++)
    {
        tmp = heads[i];

        while (tmp != NULL)
        {
            if (tmp->data == a)
                return tmp->rep;

            tmp = tmp->next;
        }
    }

    return NULL;
}


int search(int x)
{
    int i;
    struct node *tmp;

    for (i = 0; i < countRoot; i++)
    {
        tmp = heads[i];

        while (tmp != NULL)
        {
            if (tmp->data == x)
                return 1;

            tmp = tmp->next;
        }
    }

    return 0;
}


void unionSets(int a, int b)
{
    int i, j;
    struct node *rep1, *rep2, *tail2;

    rep1 = find(a);
    rep2 = find(b);

    if (rep1 == NULL || rep2 == NULL)
    {
        printf("\nElement not present in the DS\n");
        return;
    }

    if (rep1 == rep2)
    {
        printf("\nBoth elements belong to the same set\n");
        return;
    }

    for (j = 0; j < countRoot; j++)
    {
        if (heads[j] == rep2)
            break;
    }

    if (j == countRoot)
        return;

    tail2 = tails[j];

    for (i = j; i < countRoot - 1; i++)
    {
        heads[i] = heads[i + 1];
        tails[i] = tails[i + 1];
    }

    countRoot--;

    for (i = 0; i < countRoot; i++)
    {
        if (heads[i] == rep1)
        {
            tails[i]->next = rep2;
            tails[i] = tail2;
            break;
        }
    }

    while (rep2 != NULL)
    {
        rep2->rep = rep1;
        rep2 = rep2->next;
    }

    printf("\nUnion operation successful\n");
}

// Main Function
int main()
{
    int choice, x, y, i;
    struct node *rep;

    do
    {
        printf("\n-----------------------------\n");
        printf("\n.......MENU.......\n");
        printf("1. Make Set\n");
        printf("2. Display Set Representatives\n");
        printf("3. Union\n");
        printf("4. Find Set\n");
        printf("5. Exit\n");

        printf("\nEnter your choice: ");
        scanf("%d", &choice);

        printf("\n----------------------------\n");

        switch (choice)
        {
            case 1:
                printf("\nEnter new element: ");
                scanf("%d", &x);

                if (search(x))
                    printf("\nElement already present in the DS\n");
                else
                    makeSet(x);

                break;

            case 2:
                printf("\nSet Representatives:\n");

                for (i = 0; i < countRoot; i++)
                    printf("%d ", heads[i]->data);

                printf("\n");
                break;

            case 3:
                printf("\nEnter first element: ");
                scanf("%d", &x);

                printf("\nEnter second element: ");
                scanf("%d", &y);

                unionSets(x, y);
                break;

            case 4:
                printf("\nEnter the element: ");
                scanf("%d", &x);

                rep = find(x);

                if (rep == NULL)
                    printf("\nElement not present in the DS\n");
                else
                    printf("\nThe representative of %d is %d\n",
                           x, rep->data);

                break;

            case 5:
                printf("\nExiting...\n");
                return 0;

            default:
                printf("\nWrong choice\n");
                break;
        }

    } while (1);

    return 0;
}
