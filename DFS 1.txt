



#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#define MAX 10

int vertex_count = 0;

struct vertex
{
    char data;
    bool visited;
};

struct vertex *graph[MAX];
int adj_matrix[MAX][MAX];


int stack[MAX];
int top = -1;


void push(int data)
{
    if (top == MAX - 1)
    {
        printf("\nStack Overflow\n");
        return;
    }

    stack[++top] = data;
}


int pop()
{
    return stack[top--];
}


int peek()
{
    return stack[top];
}


bool is_stack_empty()
{
    return top == -1;
}


void add_vertex(char data)
{
    struct vertex *new_node;

    new_node = (struct vertex *)malloc(sizeof(struct vertex));

    if (new_node == NULL)
    {
        printf("\nMemory allocation failed\n");
        exit(1);
    }

    new_node->data = data;
    new_node->visited = false;

    graph[vertex_count] = new_node;
    vertex_count++;
}


void add_edge(int start, int end)
{
    adj_matrix[start][end] = 1;
    adj_matrix[end][start] = 1;
}


int adj_vertex(int vertex_get)
{
    int i;

    for (i = 0; i < vertex_count; i++)
    {
        if (adj_matrix[vertex_get][i] == 1 &&
            graph[i]->visited == false)
        {
            return i;
        }
    }

    return -1;
}


void display_vertex(int pos)
{
    printf("%c -> ", graph[pos]->data);
}


void dfs(int start)
{
    int unvisited;

    if (start < 0 || start >= vertex_count)
    {
        printf("\nInvalid starting vertex!\n");
        return;
    }

    top = -1;

    printf("\nDFS Traversal Order:\n");

    graph[start]->visited = true;
    display_vertex(start);
    push(start);

    while (!is_stack_empty())
    {
        unvisited = adj_vertex(peek());

        if (unvisited == -1)
        {
            pop();
        }
        else
        {
            graph[unvisited]->visited = true;
            display_vertex(unvisited);
            push(unvisited);
        }
    }

    printf("END\n");

    
    for (int i = 0; i < vertex_count; i++)
    {
        graph[i]->visited = false;
    }
}


void show()
{
    printf("\n--- Vertex Index Reference ---\n");

    for (int i = 0; i < vertex_count; i++)
    {
        printf("Position %d : '%c'\n", i, graph[i]->data);
    }

    printf("------------------------------\n");
}


int main()
{
    int num_vertices, num_edges;
    char data;
    int edge_1, edge_2, start;

    
    for (int i = 0; i < MAX; i++)
    {
        for (int j = 0; j < MAX; j++)
        {
            adj_matrix[i][j] = 0;
        }
    }

    
    printf("Enter the number of vertices (max %d): ", MAX);
    scanf("%d", &num_vertices);

    if (num_vertices <= 0 || num_vertices > MAX)
    {
        printf("Invalid number of vertices!\n");
        return 1;
    }

    for (int i = 0; i < num_vertices; i++)
    {
        printf("Enter label for vertex %d: ", i);
        scanf(" %c", &data);

        add_vertex(data);
    }

    show();

    
    printf("\nEnter the number of edges: ");
    scanf("%d", &num_edges);

    if (num_edges < 0 || num_edges > MAX * (MAX - 1) / 2)
    {
        printf("Invalid number of edges!\n");
        return 1;
    }

    for (int i = 0; i < num_edges; i++)
    {
        printf("\nEdge %d:\n", i + 1);

        printf("Enter starting vertex position (0 to %d): ",
               vertex_count - 1);
        scanf("%d", &edge_1);

        printf("Enter ending vertex position (0 to %d): ",
               vertex_count - 1);
        scanf("%d", &edge_2);

        if (edge_1 < 0 || edge_1 >= vertex_count ||
            edge_2 < 0 || edge_2 >= vertex_count)
        {
            printf("Invalid vertex position! Enter again.\n");
            i--;
        }
        else
        {
            add_edge(edge_1, edge_2);
        }
    }

    
    printf("\nEnter starting vertex position (0 to %d): ",
           vertex_count - 1);
    scanf("%d", &start);

    dfs(start);

   
    for (int i = 0; i < vertex_count; i++)
    {
        free(graph[i]);
    }

    return 0;
}
